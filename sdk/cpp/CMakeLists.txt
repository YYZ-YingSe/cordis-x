cmake_minimum_required(VERSION 3.20)
project(cordis_cpp LANGUAGES C CXX)

include(cmake/Dependencies.cmake)
if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
  add_compile_options(-fwasm-exceptions -mllvm -wasm-use-legacy-eh=false)
  add_compile_definitions(_WASI_EMULATED_SIGNAL _WASI_EMULATED_MMAN)
endif()
set(Cordis_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/include")
set(Cordis_THIRD_PARTY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party")
get_filename_component(Cordis_PROTOCOL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../protocols" ABSOLUTE)
set(Cordis_COMPONENT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/component")
include(cmake/Component.cmake)
set(CORDIS_PROTO_OUT ${Cordis_PROTOCOL_DIR}/generated/runtime/cpp)
add_library(cordis_messages STATIC ${CORDIS_PROTO_OUT}/cordis/options.pb.cc ${CORDIS_PROTO_OUT}/cordis/types.pb.cc ${CORDIS_PROTO_OUT}/transport.pb.cc)
set_target_properties(cordis_messages PROPERTIES POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(cordis_messages PUBLIC "${CORDIS_PROTO_OUT}")
target_link_libraries(cordis_messages PUBLIC protobuf::libprotobuf)
target_compile_features(cordis_messages PUBLIC cxx_std_20)

add_library(cordis INTERFACE)
add_library(cordis::core ALIAS cordis)
target_link_libraries(cordis INTERFACE cordis_messages)
target_compile_features(cordis INTERFACE cxx_std_20)
set_target_properties(cordis PROPERTIES
  CORDIS_PROTOCOL_DIR "${Cordis_PROTOCOL_DIR}"
  CORDIS_COMPONENT_SOURCE_DIR "${Cordis_COMPONENT_SOURCE_DIR}")
target_include_directories(cordis INTERFACE "${Cordis_INCLUDE_DIR}")
target_include_directories(cordis SYSTEM INTERFACE "${Cordis_THIRD_PARTY_DIR}")

if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
add_library(cordis_abi INTERFACE)
add_library(cordis::abi ALIAS cordis_abi)
target_link_libraries(cordis_abi INTERFACE cordis)
target_sources(cordis_abi INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src/abi_protobuf.cpp")
target_include_directories(cordis_abi INTERFACE "${Cordis_PROTOCOL_DIR}/abi/include")
target_compile_options(cordis_abi INTERFACE
  $<$<AND:$<COMPILE_LANGUAGE:CXX>,$<NOT:$<CXX_COMPILER_ID:MSVC>>>:-fvisibility=hidden;-fvisibility-inlines-hidden>
  $<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CXX_COMPILER_ID:GNU>>:-fno-gnu-unique>)
# Keep the complete C++ implementation private even when generated message
# objects were compiled by a consumer without hidden visibility.
if(APPLE)
  target_link_options(cordis_abi INTERFACE "LINKER:-exported_symbol,_cordis_abi_*")
elseif(UNIX)
  target_link_options(cordis_abi INTERFACE
    "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/cmake/abi-exports.map")
endif()

find_package(Threads REQUIRED)
target_link_libraries(cordis_abi INTERFACE Threads::Threads)

if(NOT WIN32)
  add_library(cordis_process INTERFACE)
  add_library(cordis::process ALIAS cordis_process)
  target_link_libraries(cordis_process INTERFACE cordis Threads::Threads)
endif()

endif()

  include("${CMAKE_CURRENT_SOURCE_DIR}/../../tests/build.cmake")

    if(WIN32)
      message(FATAL_ERROR "The process adapter requires POSIX sockets")

    include(cmake/Process.cmake)

    target_link_libraries(cordis_portable_process PRIVATE cordis::process)
    add_executable(cordis_resources_process examples/resources/process.cpp)
    target_link_libraries(cordis_resources_process PRIVATE cordis::process)
  endif()
  if(CORDIS_BUILD_ABI)
    include(cmake/Abi.cmake)
    add_library(cordis_portable_abi SHARED examples/portable/abi.cpp)
    target_link_libraries(cordis_portable_abi PRIVATE cordis::abi)
    set_target_properties(cordis_portable_abi PROPERTIES
      CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    foreach(kind counter client)
      add_library(cordis_resource_${kind} SHARED examples/resources/${kind}.cpp)
      target_link_libraries(cordis_resource_${kind} PRIVATE cordis::abi)
      set_target_properties(cordis_resource_${kind} PROPERTIES
        CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    endforeach()
  endif()
endif()
